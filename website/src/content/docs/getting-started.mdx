        // تصفية القائمة
        function filterTable() {
            const statusFilter = document.getElementById('status').value;
            const locationFilter = document.getElementById('location').value;
            const problemFilter = document.getElementById('problem').value;
            const dateFilter = document.getElementById('date').value;
            
            const filteredTasks = tasks.filter(task => {
                const statusMatch = statusFilter === 'all' || task.status === statusFilter;
                const locationMatch = locationFilter === 'all' || task.location === locationFilter;
                const problemMatch = problemFilter === 'all' || task.problem === problemFilter;
                const dateMatch = !dateFilter || task.date.includes(dateFilter);
                
                return statusMatch && locationMatch && problemMatch && dateMatch;
            });
            
            renderTaskList(filteredTasks);
        }

        function updatePerformanceIndicators() {
            // حساب عدد كل نوع من المشاكل
            const brokenLines = tasks.filter(t => t.problem.includes('خط مكسور')).length;
            const cloggedLines = tasks.filter(t => t.problem.includes('تحشير')).length;
            const meterIssues = tasks.filter(t => t.problem.includes('عداد')).length;
            
            const totalTasks = tasks.length;
            
            // حساب النسب المئوية
            const brokenLinesPercentage = totalTasks > 0 ? Math.round((brokenLines / totalTasks) * 100) : 0;
            const cloggedLinesPercentage = totalTasks > 0 ? Math.round((cloggedLines / totalTasks) * 100) : 0;
            const meterIssuesPercentage = totalTasks > 0 ? Math.round((meterIssues / totalTasks) * 100) : 0;
            
            // تحديث عرض المؤشرات
            document.getElementById('broken-lines-percentage').textContent = `${brokenLinesPercentage}%`;
            document.getElementById('clogged-lines-percentage').textContent = `${cloggedLinesPercentage}%`;
            document.getElementById('meter-issues-percentage').textContent = `${meterIssuesPercentage}%`;
            
            // تحديث شريط المؤشر
            document.getElementById('broken-lines-bar').style.width = `${brokenLinesPercentage}%`;
            document.getElementById('clogged-lines-bar').style.width = `${cloggedLinesPercentage}%`;
            document.getElementById('meter-issues-bar').style.width = `${meterIssuesPercentage}%`;
            
            // تحديث نص المؤشر
            document.getElementById('broken-lines-text').textContent = `مؤشر الأداء: ${brokenLinesPercentage}%`;
            document.getElementById('clogged-lines-text').textContent = `مؤشر الأداء: ${cloggedLinesPercentage}%`;
            document.getElementById('meter-issues-text').textContent = `مؤشر الأداء: ${meterIssuesPercentage}%`;
        }
        
        // تحديث عداد الإشعارات
        function updateNotificationBadge() {
            // حساب عدد المهام المتأخرة أكثر من 3 أيام
            const delayedTasks = tasks.filter(task => {
                const taskDate = new Date(task.date);
                const currentDate = new Date();
                const diffTime = Math.abs(currentDate - taskDate);
                const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
                return diffDays > 3 && task.status === 'pending';
            });
            
            const badge = document.getElementById('notification-badge');
            badge.textContent = delayedTasks.length;
            
            // إخفاء أو إظهار العلامة حسب عدد المهام
            if (delayedTasks.length > 0) {
                badge.style.display = 'flex';
            } else {
                badge.style.display = 'none';
            }
        }

        function resetAllData() {
            if (!db) {
                showConnectionStatus('❌ قاعدة البيانات غير مهيأة', false);
                return;
            }
            
            if (confirm('هل أنت متأكد من إعادة تعيين جميع البيانات؟ هذا الإجراء لا يمكن التراجع عنه!')) {
                db.collection('tasks').get()
                    .then(querySnapshot => {
                        const batch = db.batch();
                        querySnapshot.forEach(doc => {
                            batch.delete(doc.ref);
                        });
                        
                        return batch.commit();
                    })
                    .then(() => {
                        showConnectionStatus('✅ تمت إعادة تعيين البيانات بنجاح', true);
                        loadTasks();
                    })
                    .catch(error => {
                        console.error('Error resetting data:', error);
                        showConnectionStatus(`❌ خطأ في إعادة تعيين البيانات: ${error.message}`, false);
                    });
            }
        }
